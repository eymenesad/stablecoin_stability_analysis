% Slide 2: Project Overview
\begin{frame}{Project Overview}
\begin{block}{Objective}
Analyze stablecoin stability and depeg risk with ML models that use market data and BTC/ETH context.
\end{block}

\vspace{0.5cm}

\textbf{Scope:}
\begin{itemize}
    \item \textbf{Assets processed:} USDT (Coinbase/Kraken), USDC (Kraken), DAI (Coinbase/Kraken), TUSD, FDUSD, USDP, USDE (Binance). BUSD/FRAX/GUSD skipped (too few rows).
    \item \textbf{Data:} \(\sim\)2 years of hourly OHLCV, normalized to USD for USDT-quoted pairs, + BTC/ETH for context.
    \item \textbf{Focus:} Deviation magnitude, volatility, volume trends, correlations, and short-horizon (1h) forecasting.
\end{itemize}
\end{frame}

% Slide 3: Methodology
\begin{frame}{Methodology}
\begin{columns}[t]
\column{0.5\textwidth}
\textbf{Data Pipeline:}
\begin{itemize}
    \item \textbf{Collection:} Automated ccxt fetch (\texttt{src/data\_collector.py}).
    \item \textbf{Preprocessing:} USD normalization for USDT pairs, rolling volatility/volume, BTC/ETH merge, multi-hour lags, future labels (\texttt{src/preprocessor.py}).
\end{itemize}

\column{0.5\textwidth}
\textbf{Modeling Approach (with Comparisons):}
\begin{enumerate}
    \item \textbf{Clustering:} K-Means vs DBSCAN.
    \item \textbf{Classification:} Random Forest vs Gradient Boosting.
    \item \textbf{Anomaly:} Isolation Forest vs LOF.
    \item \textbf{Regression:} Ridge vs Neural Network.
\end{enumerate}
\end{columns}
\end{frame}

% Slide 4: Current Progress
\begin{frame}{Current Progress}
\begin{itemize}
    \item[\checkmark] \textbf{Data collection}: 2Y hourly OHLCV across target pairs.
    \item[\checkmark] \textbf{Feature set}: Volatility/volume lags (1--24h), BTC/ETH context, future labels (+1h).
    \item[\checkmark] \textbf{Models}: K-Means, Isolation Forest, RF classifier, Ridge/MLP regressors.
    \item[\checkmark] \textbf{Evaluation}: Per-symbol chronological split; predictive results saved to \texttt{reports/metrics/modeling\_results\_predictive.csv}.
\end{itemize}
\end{frame}

% Slide 11: Challenges & Next Steps
\begin{frame}{Challenges \& Next Steps}
\begin{columns}[t]
\column{0.5\textwidth}
\textbf{Challenges:}
\begin{itemize}
    \item \textbf{Imbalance:} Low recall improved to 0.42 but Low coverage remains small.
    \item \textbf{Small series:} Some assets still less than 600 rows (BUSD/FRAX/GUSD skipped); USDE/USDC/DAI\_Kraken remain thin with weak regression fit.
    \item \textbf{Anomaly precision:} ~0.46; needs duration/volume filters to curb false positives.
\end{itemize}

\column{0.5\textwidth}
\textbf{Next Steps:}
\begin{enumerate}
    \item Expand Low coverage (longer history / targeted depeg windows); tune threshold for higher Low recall.
    \item Add duration/volume spike post-filters to anomaly flags to lift precision.
    \item Consider per-symbol regression models for small/flat series; keep NN + Ridge plots.
    \item Optionally lower the row cutoff further or backfill FRAX/GUSD/BUSD to include them.
\end{enumerate}
\end{columns}
\end{frame}
